get_filename_component(panel_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(panel_sources ${openxr_panel_sources})
list(TRANSFORM panel_sources PREPEND "${panel_root}/")
add_library(Fo4VRUpscalerOpenXR SHARED ${panel_sources})
target_compile_features(Fo4VRUpscalerOpenXR PRIVATE cxx_std_23)
target_include_directories(Fo4VRUpscalerOpenXR PRIVATE "${panel_root}/src" "${panel_root}/external/openxr/include")
target_compile_definitions(Fo4VRUpscalerOpenXR PRIVATE XR_NO_PROTOTYPES XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Fo4VRUpscalerOpenXR PRIVATE /W4 /WX /EHsc /utf-8 /Zi /permissive-)
target_link_options(Fo4VRUpscalerOpenXR PRIVATE /DEBUG:FULL)
target_link_libraries(Fo4VRUpscalerOpenXR PRIVATE d3d11)
set_target_properties(Fo4VRUpscalerOpenXR PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>" PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
